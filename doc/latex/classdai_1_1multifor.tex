\doxysubsection{Detailed Description}
multifor makes it easy to perform a dynamic number of nested {\ttfamily for} loops. 

An example of the usage is as follows\+: 
\begin{DoxyCode}{0}
\DoxyCodeLine{std::vector<size\_t>\ ranges;}
\DoxyCodeLine{ranges.push\_back(\ 3\ );}
\DoxyCodeLine{ranges.push\_back(\ 4\ );}
\DoxyCodeLine{ranges.push\_back(\ 5\ );}
\DoxyCodeLine{\textcolor{keywordflow}{for}(\ \mbox{\hyperlink{classdai_1_1multifor_afe6bf38a8a7946632ec190368b19762f}{multifor}}\ s(ranges);\ s.valid();\ ++s\ )}
\DoxyCodeLine{\ \ \ \ cout\ <<\ \textcolor{stringliteral}{"{}linear\ index:\ "{}}\ <<\ (\textcolor{keywordtype}{size\_t})s\ <<\ \textcolor{stringliteral}{"{}\ corresponds\ to\ indices\ "{}}\ <<\ s[2]\ <<\ \textcolor{stringliteral}{"{},\ "{}}\ <<\ s[1]\ <<\ \textcolor{stringliteral}{"{},\ "{}}\ <<\ s[0]\ <<\ endl;}

\end{DoxyCode}
 which would be equivalent to\+: 
\begin{DoxyCode}{0}
\DoxyCodeLine{\textcolor{keywordtype}{size\_t}\ s\ =\ 0;}
\DoxyCodeLine{\textcolor{keywordflow}{for}(\ \textcolor{keywordtype}{size\_t}\ s2\ =\ 0;\ s2\ <\ 5;\ s2++\ )}
\DoxyCodeLine{\ \ \ \ \textcolor{keywordflow}{for}(\ \textcolor{keywordtype}{size\_t}\ s1\ =\ 0;\ s1\ <\ 4;\ s1++\ )}
\DoxyCodeLine{\ \ \ \ \ \ \ \ \textcolor{keywordflow}{for}(\ \textcolor{keywordtype}{size\_t}\ s0\ =\ 0;\ s0\ <\ 3;\ s++,\ s0++\ )}
\DoxyCodeLine{\ \ \ \ \ \ \ \ \ \ \ \ cout\ <<\ \textcolor{stringliteral}{"{}linear\ index:\ "{}}\ <<\ (\textcolor{keywordtype}{size\_t})s\ <<\ \textcolor{stringliteral}{"{}\ corresponds\ to\ indices\ "{}}\ <<\ s2\ <<\ \textcolor{stringliteral}{"{},\ "{}}\ <<\ s1\ <<\ \textcolor{stringliteral}{"{},\ "{}}\ <<\ s0\ <<\ endl;}

\end{DoxyCode}
