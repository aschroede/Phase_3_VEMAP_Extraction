458 template<typename binOp> TFactor<T> binaryTr( const TFactor<T> &g, binOp op ) const {
459 // Note that to prevent a copy to be made, it is crucial
460 // that the result is declared outside the if-else construct.
461
462 bool instantiationDataOnRight = !g._i.i().front().empty();
463
464 TFactor<T> result;
465 if( _vs == g._vs ) { // optimize special case
466 result._vs = _vs;
467 result._p = _p.pwBinaryTr( g._p, op );
468
469 if (instantiationDataOnRight == true)
470 result._i = _i.pwBinaryTr( g._i );
471 else
472 result._i = _i;
473 } else {
474 result._vs = _vs | g._vs;
475 size_t N = BigInt_size_t( result._vs.nrStates() );
476
477 IndexFor i_f( _vs, result.vars() );
478 IndexFor i_g( g._vs, result.vars() );
479
480 result._p.p().clear();
481 result._p.p().reserve( N );
482 result._i.i().clear();
483 result._i.i().reserve( N );
484
485 for( size_t i = 0; i < N; i++, ++i_f, ++i_g ){
486 result._p.p().push_back( op( _p[i_f], g[i_g] ) );
487
488 // If right factor contains instantiation data then copy it over
489 if (instantiationDataOnRight == true) {
490 result._i.i().push_back( g._i[i_g]);
491 }
492
493 // Else left factor has instantiation data to copy over
494 else{
495 result._i.i().push_back( _i[i_f]);
496 }
497 }
498 }
499 return result;
500 }

613template<typename T> TFactor<T> TFactor<T>::maxMarginalTransparent(const VarSet &vars, bool normed) const {
614
615 // Get the intersection of the input vars (those to not maximise out) and the vars in the factor _vs
616 // In this case it would be {1} intersect {0, 1, 2, 3, 4} = {1}
617 // Residual vars are those that remain after maximizing out (the ones to keep)
618
619 if(vars == _vs){
620 return this->normalized();
621 }
622
623
624 dai::VarSet res_vars = vars & _vs;
625 dai::VarSet to_max_out = _vs / vars;
626 Var varToMaxOut = to_max_out.front();
627
628 // Create a new factor "res" that contains the residual variables and set all probabilities to 0.0
629 TFactor<T> res( res_vars, 0.0 );
630
631 // The class IndexFor is an important tool for indexing Factor entries.
632 // Then the following code: loops over all joint states of the variables in _vs,
633 // and (size_t)i equals the linear index of the corresponding state of res_vars,
634 // where the variables in res_vars that are not in forVars assume their zero'th value.
635 IndexFor i_res( res_vars, _vs );
636
637 State S(_vs);
638 // Loop over all entries in the factor's probability table '_p'.
639 for( size_t i = 0; i < _p.size(); i++, ++i_res, S++){
640
641 // If the current entry in '_p' is greater than the corresponding entry in 'res', update 'res'.
642 if( _p[i] > res._p[i_res] ){
643 res.set( i_res, _p[i] );
644 std::map<Var, size_t> rowInstantiation = getInstantiation( i_res );
645
646 auto it = rowInstantiation.find(varToMaxOut);
647
648 if (it != rowInstantiation.end()) {
649 // Key exists, update the value
650 it->second = S(varToMaxOut);
651 } else {
652 // Key doesn't exist, add a new key-value pair
653 rowInstantiation[varToMaxOut] = S(varToMaxOut);
654 }
655
656 res.setInstantiation( i_res, rowInstantiation);
657 //res.setInstantiation(to_max_out.front(), S(to_max_out.front()));
658 //_instantiation.push_back(std::make_pair(to_max_out.front(), S(to_max_out.front())));
659 }
660 }
661
662 if( normed )
663 res.normalize( NORMPROB );
664
665 // Return the resulting factor.
666 return res;
667

670 // in a varset to it's state

691template<typename T> T TFactor<T>::strength( const Var &i, const Var &j ) const {
692 DAI_DEBASSERT( _vs.contains( i ) );
693 DAI_DEBASSERT( _vs.contains( j ) );
694 DAI_DEBASSERT( i != j );
695 VarSet ij(i, j);
696
697 T max = 0.0;
698 for( size_t alpha1 = 0; alpha1 < i.states(); alpha1++ )
699 for( size_t alpha2 = 0; alpha2 < i.states(); alpha2++ )
700 if( alpha2 != alpha1 )
701 for( size_t beta1 = 0; beta1 < j.states(); beta1++ )
702 for( size_t beta2 = 0; beta2 < j.states(); beta2++ )
703 if( beta2 != beta1 ) {
704 size_t as = 1, bs = 1;
705 if( i < j )
706 bs = i.states();
707 else
708 as = j.states();
709 T f1 = slice( ij, alpha1 * as + beta1 * bs ).p().divide( slice( ij, alpha2 * as + beta1 * bs ).p() ).max();
710 T f2 = slice( ij, alpha2 * as + beta2 * bs ).p().divide( slice( ij, alpha1 * as + beta2 * bs ).p() ).max();
711 T f = f1 * f2;
712 if( f > max )
713 max = f;
714 }
715
716 return std::tanh( 0.25 * std::log( max ) );
717}

770template<typename T> T MutualInfo(const TFactor<T> &f) {
771 DAI_ASSERT( f.vars().size() == 2 );
772 VarSet::const_iterator it = f.vars().begin();
773 Var i = *it; it++; Var j = *it;
774 TFactor<T> projection = f.marginal(i) * f.marginal(j);
